cmake_minimum_required(VERSION 3.24)
project(driver-health-auditor VERSION 1.0.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Windows and MSVC-compatible compiler required")
endif()
option(DHA_ANALYZE "Enable MSVC static analysis" OFF)
option(DHA_ASAN "Enable MSVC AddressSanitizer" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(dha_core
  src/model.cpp src/cli.cpp src/correlation.cpp src/audit.cpp src/findings.cpp
  src/json_formatter.cpp src/human_formatter.cpp src/pnp.cpp src/package.cpp
  src/service.cpp src/local_paths.cpp src/signature.cpp src/pe_inspection.cpp)
target_include_directories(dha_core PUBLIC include)
target_compile_definitions(dha_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_compile_options(dha_core PUBLIC /W4 /WX /utf-8 /permissive- /EHsc)
target_link_libraries(dha_core PUBLIC setupapi cfgmgr32 advapi32 wintrust crypt32)
if(DHA_ANALYZE)
  target_compile_options(dha_core PUBLIC /analyze)
endif()
if(DHA_ASAN)
  target_compile_options(dha_core PUBLIC /fsanitize=address /Zi)
  target_link_options(dha_core PUBLIC /INCREMENTAL:NO)
  get_filename_component(dha_compiler_dir "${CMAKE_CXX_COMPILER}" DIRECTORY)
  if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    set(dha_asan_arch x86_64)
  else()
    set(dha_asan_arch i386)
  endif()
  set(dha_asan_runtime "${dha_compiler_dir}/clang_rt.asan_dynamic-${dha_asan_arch}.dll")
  if(NOT EXISTS "${dha_asan_runtime}")
    message(FATAL_ERROR "MSVC ASan runtime not found beside compiler")
  endif()
endif()
add_executable(driver-health-auditor src/main.cpp app.manifest)
target_link_libraries(driver-health-auditor PRIVATE dha_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(dha_tests tests/tests.cpp)
  target_link_libraries(dha_tests PRIVATE dha_core)
  add_test(NAME synthetic COMMAND dha_tests)
  add_test(NAME help COMMAND driver-health-auditor --help)
  add_test(NAME version COMMAND driver-health-auditor --version)
  set_tests_properties(synthetic help version PROPERTIES TIMEOUT 30)
  add_executable(dha_nonadmin tests/nonadmin.cpp)
  target_compile_definitions(dha_nonadmin PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(dha_nonadmin PRIVATE advapi32)
  target_compile_options(dha_nonadmin PRIVATE /W4 /WX /utf-8)
endif()
if(DHA_ASAN)
  set(dha_instrumented driver-health-auditor)
  if(BUILD_TESTING)
    list(APPEND dha_instrumented dha_tests)
  endif()
  foreach(dha_target IN LISTS dha_instrumented)
    add_custom_command(TARGET ${dha_target} POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${dha_asan_runtime}" "$<TARGET_FILE_DIR:${dha_target}>"
      VERBATIM)
  endforeach()
endif()
install(TARGETS driver-health-auditor RUNTIME DESTINATION bin)
install(FILES README.md LICENSE VALIDATION.md DESTINATION share/driver-health-auditor)
install(FILES docs/json-schema.json DESTINATION share/driver-health-auditor/docs)
